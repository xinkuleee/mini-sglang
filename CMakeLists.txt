cmake_minimum_required(VERSION 3.24)
project(mini_sglang_cpp VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MINISGL_WITH_LLAMA "Build the real GGUF inference engine and CLI" OFF)
option(MINISGL_CUDA "Enable llama.cpp CUDA kernels" OFF)
option(MINISGL_BUILD_TESTS "Build deterministic core tests" ON)
set(MINISGL_LLAMA_SOURCE "" CACHE PATH "Local pinned llama.cpp checkout")
set(MINISGL_HTTPLIB_SOURCE "" CACHE PATH "Local cpp-httplib v0.20.0 checkout")
set(MINISGL_JSON_SOURCE "" CACHE PATH "Local nlohmann/json v3.12.0 checkout")

add_library(minisgl_core STATIC src/types.cpp src/sampling.cpp src/cache.cpp src/scheduler.cpp)
target_include_directories(minisgl_core PUBLIC include)
target_compile_features(minisgl_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(minisgl_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(minisgl_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(MINISGL_BUILD_TESTS)
  enable_testing()
  add_executable(minisgl-core-tests tests/core_tests.cpp)
  target_link_libraries(minisgl-core-tests PRIVATE minisgl_core)
  add_test(NAME minisgl-core COMMAND minisgl-core-tests)
endif()

if(MINISGL_WITH_LLAMA)
  add_subdirectory(native)
  add_library(minisgl_llama STATIC src/llama_engine.cpp)
  target_link_libraries(minisgl_llama PUBLIC minisgl_core PRIVATE minisgl_bridge)
  target_compile_definitions(minisgl_llama PUBLIC MINISGL_WITH_LLAMA=1)

  include(FetchContent)
  if(NOT MINISGL_HTTPLIB_SOURCE)
    FetchContent_Declare(minisgl_httplib
      GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
      GIT_TAG 787a34ad7f01f20922a237d5142aae469828be72 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(minisgl_httplib)
    set(MINISGL_HTTPLIB_SOURCE "${minisgl_httplib_SOURCE_DIR}")
  endif()
  if(NOT MINISGL_JSON_SOURCE)
    FetchContent_Declare(minisgl_json
      GIT_REPOSITORY https://github.com/nlohmann/json.git
      GIT_TAG 55f93686c01528224f448c19128836e7df245f72 GIT_SHALLOW FALSE)
    set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(minisgl_json)
    set(MINISGL_JSON_SOURCE "${minisgl_json_SOURCE_DIR}")
  endif()
  if(NOT EXISTS "${MINISGL_HTTPLIB_SOURCE}/httplib.h" OR NOT EXISTS "${MINISGL_JSON_SOURCE}/single_include/nlohmann/json.hpp")
    message(FATAL_ERROR "HTTP dependency source paths do not contain the expected headers")
  endif()
  find_package(Threads REQUIRED)
  add_executable(minisgl-cli src/main.cpp src/server.cpp)
  set_target_properties(minisgl-cli PROPERTIES OUTPUT_NAME mini-sglang)
  target_include_directories(minisgl-cli PRIVATE "${MINISGL_HTTPLIB_SOURCE}" "${MINISGL_JSON_SOURCE}/single_include")
  target_link_libraries(minisgl-cli PRIVATE minisgl_llama Threads::Threads)
  if(WIN32)
    target_link_libraries(minisgl-cli PRIVATE ws2_32 crypt32)
  endif()
  if(MINGW)
    target_link_options(minisgl-cli PRIVATE -static-libgcc -static-libstdc++)
  endif()
  if(MSVC)
    target_compile_options(minisgl-cli PRIVATE /utf-8)
  endif()
endif()
