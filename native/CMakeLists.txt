cmake_minimum_required(VERSION 3.24)
project(minisgl_native LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MINISGL_BRIDGE_SHARED "Build a single FFI dynamic library" OFF)
option(MINISGL_CUDA "Enable the llama.cpp CUDA backend" OFF)
option(MINISGL_CPU_BASELINE "Use the portable baseline CPU instruction set" ON)
option(MINISGL_NATIVE_TESTS "Build native ABI smoke executable" OFF)
set(MINISGL_LLAMA_SOURCE "" CACHE PATH "Local llama.cpp checkout at the pinned revision")
set(MINISGL_LLAMA_COMMIT "64e9bceb2c3a856efed96feda784a50947049feb")

# 核心库静态链接进唯一 adapter DLL，Rust 不需要枚举 llama/ggml 的传递链接项。
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(MINISGL_CPU_BASELINE)
  foreach(extension SSE42 AVX AVX2 BMI2 FMA F16C AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16)
    set(GGML_${extension} OFF CACHE BOOL "Portable CPU baseline" FORCE)
  endforeach()
else()
  foreach(extension SSE42 AVX AVX2 BMI2 FMA F16C)
    set(GGML_${extension} ON CACHE BOOL "llama.cpp optimized x86 defaults" FORCE)
  endforeach()
endif()
set(GGML_CUDA ${MINISGL_CUDA} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)

if(MINISGL_LLAMA_SOURCE)
  find_package(Git REQUIRED)
  execute_process(COMMAND ${GIT_EXECUTABLE} -C "${MINISGL_LLAMA_SOURCE}" rev-parse HEAD
    RESULT_VARIABLE revision_status OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT revision_status EQUAL 0 OR NOT revision STREQUAL MINISGL_LLAMA_COMMIT)
    message(FATAL_ERROR "MINISGL_LLAMA_SOURCE must be a git checkout at ${MINISGL_LLAMA_COMMIT}; got ${revision}")
  endif()
  add_subdirectory("${MINISGL_LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama.cpp" EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(llama_cpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
    GIT_TAG ${MINISGL_LLAMA_COMMIT} GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(llama_cpp)
endif()

if(MINISGL_BRIDGE_SHARED)
  add_library(minisgl_bridge SHARED minisgl_bridge.cpp)
  target_compile_definitions(minisgl_bridge PUBLIC MSGL_SHARED PRIVATE MSGL_BUILD)
else()
  add_library(minisgl_bridge STATIC minisgl_bridge.cpp)
endif()
target_include_directories(minisgl_bridge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(minisgl_bridge PRIVATE llama)
set_target_properties(minisgl_bridge PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
  ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(minisgl_bridge PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib"
    LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib"
    ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
if(MINGW AND MINISGL_BRIDGE_SHARED)
  target_link_options(minisgl_bridge PRIVATE -static-libgcc -static-libstdc++)
endif()

if(MINISGL_NATIVE_TESTS)
  enable_testing()
  add_executable(minisgl-abi-smoke abi_smoke.cpp)
  target_link_libraries(minisgl-abi-smoke PRIVATE minisgl_bridge)
  if(MINGW)
    target_link_options(minisgl-abi-smoke PRIVATE -static-libgcc -static-libstdc++)
  endif()
  set_target_properties(minisgl-abi-smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(minisgl-abi-smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
  endforeach()
  add_test(NAME native-invalid-config COMMAND minisgl-abi-smoke)
endif()
